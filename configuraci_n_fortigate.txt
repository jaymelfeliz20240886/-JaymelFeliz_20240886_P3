config system interface
    edit "port1"
        set mode dhcp
        set allowaccess ping https http ssh fgfm
    next
    edit "port2"
        set vdom "root"
        set ip 10.20.24.1 255.255.255.240
        set allowaccess ping
    next
    edit "VLAN10_LIMITADOS"
        set vdom "root"
        set ip 10.10.10.1 255.255.255.0
        set allowaccess ping
        set interface "port3"
        set vlanid 10
    next
    edit "VLAN20_SSH"
        set vdom "root"
        set ip 10.20.20.1 255.255.255.0
        set allowaccess ping ssh
        set interface "port3"
        set vlanid 20
    next
end

config system dns
    set primary 8.8.8.8
    set secondary 8.8.4.4
end

config system settings
    set gui-multiple-interface-policy enable
end

config firewall address
    edit "archive.ubuntu.com"
        set type fqdn
        set fqdn "archive.ubuntu.com"
    next
    edit "8.8.8.8"
        set type ipmask
        set subnet 8.8.8.8 255.255.255.255
    next
    edit "DMZ_Subnet"
        set subnet 10.20.24.0 255.255.255.240
    next
end

config firewall policy
    edit 1
        set name "DMZ_TO_UPDATES"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "DMZ_Subnet"
        set dstaddr "archive.ubuntu.com" "8.8.8.8"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
    edit 2
        set name "INTERNET_VLANS"
        set srcintf "VLAN10_LIMITADOS" "VLAN20_SSH"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
end
